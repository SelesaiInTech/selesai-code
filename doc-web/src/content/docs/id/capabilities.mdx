import CapabilityList from "../../../components/CapabilityList.astro";
import { categories, capabilities } from "../../../data/capabilities.ts";

Selesai mengirimkan kemampuan bundel di bawah ini sebagai bagian dari package `@selesai/code`. Ekstensi opsional dicantumkan terpisah di bagian bawah halaman. Ekstensi built-in inti didokumentasikan di halaman [Kustomisasi](/id/customization/). Setiap entri menautkan ke panduan dengan contoh, konfigurasi, batasan, dan bukti source.

{categories.map((category) => (
  <section className="category-section" id={category.id}>
    <h2>{category.indonesian}</h2>
    <CapabilityList category={category.id} locale="id" />
  </section>
))}

{capabilities.some((c) => c.distribution === "core") && (
  <section className="category-section" id="core-extensions">
    <h2>Ekstensi built-in inti</h2>
    <p>Kemampuan ini adalah bagian dari runtime Selesai dan tidak tercantum di manifest ekstensi. Lihat halaman <a href={`${import.meta.env.BASE_URL}id/customization/`}>Kustomisasi</a> untuk ikhtisar lengkap.</p>
    {categories.map((category) => (
      <CapabilityList category={category.id} locale="id" distribution="core" />
    ))}
  </section>
)}

{capabilities.some((c) => c.distribution === "optional") && (
  <section className="category-section" id="optional-extensions">
    <h2>Ekstensi opsional</h2>
    <p>Ekstensi berikut tersedia tetapi tidak dimuat otomatis. Pasang atau aktifkan secara terpisah sebelum digunakan.</p>
    {categories.map((category) => (
      <CapabilityList category={category.id} locale="id" distribution="optional" />
    ))}
  </section>
)}
